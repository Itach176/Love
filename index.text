    <script>
        // Typing animation
        const messages = [
            "I wanted to make something special for you...",
            "Because you are special to me ❤️",
            "Every moment with you feels magical ✨",
            "You always make my day special 😇",
            "I Love You So much 🤍",
            "Thank You So Much for Everything 🥺",
            "touch below button and see the magic 💖"
        

        ];

        let msgIndex = 0;
        let charIndex = 0;
        const textElement = document.getElementById("text");

        function typeText() {
            if (charIndex < messages[msgIndex].length) {
                textElement.innerHTML += messages[msgIndex].charAt(charIndex);
                charIndex++;
                setTimeout(typeText, 50);
            } else {
                setTimeout(() => {
                    textElement.innerHTML = "";
                    charIndex = 0;
                    msgIndex = (msgIndex + 1) % messages.length;
                    typeText();
                }, 2000);
            }
        }
        typeText();

        // START EXPERIENCE
        function startExperience() {
            // Display the hidden content
            document.getElementById("magic").style.display = "block";

            const music = document.getElementById("bgMusic");
            music.volume = 0.4;
            
            // Playing music via a promise to handle browser restrictions
            let playPromise = music.play();

            if (playPromise !== undefined) {
                playPromise.then(_ => {
                    console.log("Music started!");
                }).catch(error => {
                    console.log("Music failed to play. Make sure you interacted with the page first.");
                });
            }
        }

        // Show reasons
        function showReasons() {
            document.getElementById("reasons").style.display = "block";
            // Scroll to bottom so they see the new text
            window.scrollTo({ top: document.body.scrollHeight, behavior: 'smooth' });
        }

        // Floating hearts
        function createHeart() {
            const heart = document.createElement("div");
            heart.classList.add("heart");
            heart.innerHTML = "❤️";
            heart.style.left = Math.random() * 100 + "vw";
            heart.style.fontSize = Math.random() * 20 + 15 + "px";
            heart.style.opacity = Math.random();

            document.body.appendChild(heart);

            setTimeout(() => heart.remove(), 5000);
        }
        setInterval(createHeart, 300);
    </script>
